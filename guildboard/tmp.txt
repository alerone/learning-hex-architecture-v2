MARA=$(curl -s -X POST localhost:8080/api/adventurers \
-H "Authorization: Bearer $MASTER" \
-H 'Content-Type: application/json' -d '{"name":"Mara"}' | jq -r '.id')

echo $MARA

code() { curl -s -o /dev/null -w ' %{http_code}\n' "$@"; }

tok() {
curl -s -X POST localhost:9000/token \
-H 'Content-Type: application/json' \
-d "{\"username\":\"$1\",\"password\":\"$1\"}" |
grep -o '"access_token" *: *"[^"]*"' | cut -d'"' -f4
}

./gradlew :token-issuer:bootJar
docker compose up -d --build --wait token-issuer
MASTER=$(scripts/tok.sh master); HERO=$(scripts/tok.sh hero)

code localhost:8081/actuator/health
code localhost:8081/actuator/metrics
docker compose logs quests | grep 'notification to'
